option(MIR_OPTIMIZE_ASSET_COOKER "Optimize offline asset cooking" ON)
if(MSVC AND MIR_OPTIMIZE_ASSET_COOKER)
    # Offline tools keep the Debug runtime and symbols, /RTC1 cannot combine with /O2.
    separate_arguments(asset_cooker_original_debug_flags WINDOWS_COMMAND "${CMAKE_CXX_FLAGS_DEBUG}")
    string(REGEX REPLACE "/RTC1|/Ob0|/Od" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
endif()

add_executable(AssetCooker)
find_package(fastgltf CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Stb REQUIRED)
add_library(gltf_importer STATIC AssetReferences.cpp GltfImporter.cpp IdMigration.cpp ImageImporter.cpp MaterialCooker.cpp)
target_include_directories(gltf_importer PRIVATE ${Stb_INCLUDE_DIR})
target_link_libraries(gltf_importer PUBLIC mir::assets PRIVATE fastgltf::fastgltf mir::prefab mir::particles nlohmann_json::nlohmann_json)
target_sources(AssetCooker PRIVATE main.cpp)
# The compiler is a cooker dependency, game binaries load only cooked SPIR-V.
target_compile_definitions(AssetCooker PRIVATE PARTICLE_SLANGC_EXECUTABLE="${SLANGC_EXECUTABLE}")
engine_target_utf8(AssetCooker)
# terrain: only for IsTerrainFileExcludedFromCook.
target_link_libraries(AssetCooker PRIVATE gltf_importer mir::assets mir::core mir::terrain mir::prefab mir::particles nlohmann_json::nlohmann_json)
if(MSVC AND MIR_OPTIMIZE_ASSET_COOKER)
    target_compile_options(gltf_importer PRIVATE "$<$<CONFIG:Debug>:/O2>")
    target_compile_options(AssetCooker PRIVATE "$<$<CONFIG:Debug>:/O2>")
endif()

    target_include_directories(gltf_importer_tests PRIVATE ${Stb_INCLUDE_DIR})

    if(MSVC AND MIR_OPTIMIZE_ASSET_COOKER)
        # The test executable itself retains the original Debug compiler checks.

            target_compile_options(gltf_importer_tests PRIVATE "$<$<CONFIG:Debug>:${flag}>")
        endforeach()
    endif()
    target_link_libraries(gltf_importer_tests PRIVATE gltf_importer mir::animation mir::prefab mir::particles nlohmann_json::nlohmann_json Catch2::Catch2WithMain)
